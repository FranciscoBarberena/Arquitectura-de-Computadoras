negado db "Acceso denegado",10

bloqueo db "Acceso BLOQUEADO",10

LEER\_CONTRA:

push bx

push cx

inc bx

dec cl

jnz loop

pop cx

pop bx

ret

comparar\_string: mov bx,sp

push bx

add bx,2

mov bx,\[bx\]

mov ax,\[bx\]

pop bx

add bx,4

mov bx,\[bx\]

mov bx,\[bx\]

cmp bx, ax

jz iguales

mov al,0

jmp fin

iguales:

mov al,0FFh

cmp global,1

jz fin

mov al,0FFh

mov global, 1

mov bx, offset intento + 2

push bx

mov bx, offset clave + 2

call comparar\_string

pop bx

pop bx

fin: ret

mostrar\_mensaje: cmp al, 0Ffh

jnz distintos

mov bx, offset exito

mov al, offset bloqueo - offset negado
