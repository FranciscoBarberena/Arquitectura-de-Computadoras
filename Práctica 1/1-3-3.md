org 3000h

ES\_MAYUS:

mov cl, \[bx\]

cmp cl, 91

jns no

jmp seguir

no: mov ah, 0

jmp seguirr

jmp fin

seguir:

cmp cl, 64

js no

mov ah, 0FFh

cmp ah,0

jnz A\_MINUS

jmp seguirr

fin: ret

A\_MINUS:

add byte ptr\[bx\], 32

seguirr: inc bx

dec al

cmp al,0

jnz string\_a\_minus

ret

STRING\_A\_MINUS: jmp ES\_MAYUS

ret
